# What a crawler may read.
#
# This file exists because /robots.txt used to fall through to the API's
# catch-all and answer 401 UNAUTHENTICATED. Google's current rule is that a
# 4xx means "no robots.txt, no restrictions", but a 401 specifically has
# historically been read as "this whole site is private", and other crawlers
# still read it that way. A site whose front door is meant to be found should
# not leave that to interpretation.

User-agent: *

# The operator's console. Behind a sign-in, useful to nobody who is not
# already holding an account, and every crawl of it is a 401.
Disallow: /console

# The live-location pages a traveller shares in an emergency. The 32-byte
# token in the URL is the entire credential, so these must never be fetched
# by anything that was not handed the link. They already answer
# `x-robots-tag: noindex`; this stops the request from being made at all.
Disallow: /sos/

# Everything else is open on purpose - the app itself, an area's feed, the
# board, a story, and above all a statement: an evidence record nobody can
# find is not evidence (docs/31).

# THE ANSWER ENGINES, NAMED - AND THE DISALLOWS REPEATED, WHICH IS THE POINT.
#
# The group above already allows every crawler below it, so naming them grants
# nothing new. Two of these names are here because they do not govern crawling
# at all: Google-Extended and Applebot-Extended decide whether pages already
# fetched may be used to ground a generated answer, and each defaults to
# allowed only for as long as nobody has said otherwise. Writing the decision
# down is the difference between a policy and an accident.
#
# THE REPEATED Disallow LINES ARE NOT REDUNDANT. A crawler obeys exactly one
# group - its own, if it has one - and ignores the wildcard group completely.
# Naming GPTBot without repeating these two lines would hand it /sos/, whose
# URL token is the entire credential protecting somebody sharing their live
# location in an emergency. Every name added below must carry them.
#
# Saying yes is deliberate: an evidence record nobody can find is not evidence
# (docs/31), and that holds whether the reader is a person, a search engine, or
# something answering a question on their behalf.

User-agent: GPTBot
User-agent: OAI-SearchBot
User-agent: ChatGPT-User
User-agent: ClaudeBot
User-agent: Claude-SearchBot
User-agent: Claude-User
User-agent: PerplexityBot
User-agent: Perplexity-User
User-agent: Google-Extended
User-agent: Applebot-Extended
User-agent: Amazonbot
User-agent: meta-externalagent
Allow: /
Disallow: /console
Disallow: /sos/

# The promo page (/th/, and /th/en and /th/zh beside it) and the
# machine-readable summary (/llms.txt) are the pages meant to be found and
# cited, by search engines and answer engines alike.
Sitemap: https://chivargo.com/sitemap.xml
